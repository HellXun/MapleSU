cmake_minimum_required(VERSION 3.22)
project(maplesu_policy LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(SELINUX_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/selinux" CACHE PATH "Pinned SELinux source")
file(GLOB SEPOL_SOURCES CONFIGURE_DEPENDS "${SELINUX_SOURCE}/libsepol/src/*.c")
add_library(maplesu_sepol STATIC ${SEPOL_SOURCES})
target_include_directories(maplesu_sepol PUBLIC "${SELINUX_SOURCE}/libsepol/include" PRIVATE "${SELINUX_SOURCE}/libsepol/src")
target_compile_definitions(maplesu_sepol PRIVATE _GNU_SOURCE)
target_compile_options(maplesu_sepol PRIVATE -Wno-unused-result)
add_executable(maplesu-policy src/main.cpp src/jailbreak_selinux_hide.cpp src/ksu_manager_fd.cpp src/boot_sdk_control.cpp src/boot_sdk_layout.cpp derived/sepolicy.cpp
    ../runtime/ultra/policy/module_policy_config.cpp ../runtime/ultra/policy/kernel_layout.cpp
    ../runtime/ultra/policy/jailbreak_status_layout.cpp)
target_include_directories(maplesu-policy PRIVATE ../runtime/ultra/policy ../third_party/ksu-uapi)
set(MAPLESU_POLICY_HOOK_BUNDLE_SOURCE "" CACHE FILEPATH "Generated, verified temporary Hook policy bundles")
if(MAPLESU_POLICY_HOOK_BUNDLE_SOURCE)
  if(NOT EXISTS "${MAPLESU_POLICY_HOOK_BUNDLE_SOURCE}")
    message(FATAL_ERROR "Temporary Hook policy bundle source is missing")
  endif()
  target_sources(maplesu-policy PRIVATE src/policy_hook_loader.cpp ../runtime/ultra/policy/kernel_payload.cpp
    ../runtime/ultra/policy/kernel_payload_loader.cpp ../runtime/ultra/policy/kernel_payload_metadata.cpp
    ../runtime/ultra/policy/kernel_payload_hook.cpp "${MAPLESU_POLICY_HOOK_BUNDLE_SOURCE}")
  target_compile_features(maplesu-policy PRIVATE cxx_std_20)
  target_compile_definitions(maplesu-policy PRIVATE MAPLESU_POLICY_HOOK_BACKEND=1)
  target_link_libraries(maplesu-policy PRIVATE z)
endif()
add_executable(maplesu-module-policy-config-tests tests/module_policy_config_tests.cpp
    ../runtime/ultra/policy/module_policy_config.cpp)
target_include_directories(maplesu-module-policy-config-tests PRIVATE ../runtime/ultra/policy)
target_compile_options(maplesu-module-policy-config-tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_executable(maplesu-policy-tests tests/policy_tests.cpp derived/sepolicy.cpp
    ../runtime/ultra/policy/module_policy_config.cpp)
foreach(target maplesu-policy maplesu-policy-tests)
  target_include_directories(${target} PRIVATE derived src ../runtime/ultra/policy)
  target_link_libraries(${target} PRIVATE maplesu_sepol)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  if(ANDROID)
    target_link_options(${target} PRIVATE -Wl,-z,max-page-size=16384)
  endif()
endforeach()
